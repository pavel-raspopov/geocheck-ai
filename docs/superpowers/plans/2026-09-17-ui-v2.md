# UI v2 (Feature 11) Implementation Plan — task text + rules confirmation + verdict checklist

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the v1 rule-select UI with the ТЗ v2 flow: user types the task text → offline parser extracts rules → user reviews and confirms them (human-in-the-loop) → multi-rule verification runs → verdict checklist (one row per rule + aggregate). A collapsed Gemini fallback section re-extracts rules on explicit user action with a user-supplied API key.

**Architecture:** Pure formatting logic (`relation-format.ts`) is TDD'd with Vitest; DOM components (`task-text`, `rules-preview`, `gemini-fallback`, `verdict-checklist`) are vanilla function factories following the existing `src/ui/*` pattern and are verified via the headless-QA harness (`scripts/qa/ui-shell.mjs`). `run.ts` gains `analyzeImage()` (analysis without the single-rule `verify`), so `app.ts` composes `analyzeImage` → `evaluateRules(graph, confirmed.relations, ε)`. Demo mode evaluates the same confirmed rules against the demo graph.

**Tech Stack:** Vanilla TS + Vite SPA, Vitest (TDD for pure logic), no frameworks, no Tailwind, no backend.

## Global Constraints

- TypeScript strict (`noUncheckedIndexedAccess`, `exactOptionalPropertyTypes`); no `any`.
- Thresholds stay in `src/pipeline/constants.ts` — never inline pipeline numbers.
- Design tokens from `context/ui-tokens.md` (CSS variables in `src/styles.css`) — no raw hex in components.
- UI copy in Russian. ТЗ message strings are contract (already produced by `rules-engine.ts` — do not touch them).
- Human-in-the-loop gate: verification runs ONLY from the «Подтвердить и проверить» button. Editing the text invalidates the confirmation.
- Gemini is invoked ONLY by explicit user click («Уточнить через ИИ»). API key lives at most in `localStorage` (key `geocheck.gemini-api-key`); never in git, never in `.env`, never logged.
- Secrets: never write any real API key into any file, test, or doc. The live CORS check (Task 8) uses a key the USER types into the running UI.
- **Project commit rule (overrides per-task commits):** TDD red/green cycles stay uncommitted; the whole feature is delivered as ONE commit `feat(ui): task-text workflow with rule confirmation and verdict checklist (step 11)` at Task 9, after `/remember save`.
- Windows/PowerShell: use `pnpm.cmd`; gate exit codes via `cmd /c "pnpm <cmd> && echo PASS || echo FAIL"`; never `2>&1` on native commands.
- Verification ladder before "done": `pnpm test` + `pnpm typecheck` + `pnpm lint` → `pnpm build` + `pnpm preview` smoke → live headless QA (`pnpm qa`).

/** Человекочитаемые RU-строки для распознанных правил (фича 11: предпросмотр + чеклист). */
import type { ParsedTask, Relation } from '../pipeline/rules/types';

/** Одно правило → строка предпросмотра («∠ABC = 90°», «BK — медиана к стороне AC»). */
export function formatRelation(r: Relation): string {
  switch (r.kind) {
    case 'angle':
      return `∠${r.angle} = ${r.degrees}°`;
    case 'parallel':
      return `${r.a} ∥ ${r.b}`;
    case 'equal':
      return `|${r.a}| = |${r.b}|`;
    case 'on-segment':
      return `${r.point} ∈ ${r.segment}`;
    case 'median':
      return `${r.cevian} — медиана к стороне ${r.side}`;
    case 'bisector':
      return `${r.cevian} — биссектриса угла ${r.angle}`;
    case 'height':
      return `${r.cevian} — высота к стороне ${r.side}`;
  }
}

export interface TaskSummary {
  readonly rules: string[];
  readonly givens: string[];
}

/** ParsedTask → списки строк для UI (правила + «дано»). */
export function formatTaskSummary(task: ParsedTask): TaskSummary {
  return { rules: task.relations.map(formatRelation), givens: [...task.givens] };
}
```

/** Результат анализа без вердикта: вход для мульт-проверки `evaluateRules` (фича 11). */
export type AnalyzeResult = Omit<PipelineResult, 'verdict'>;

/**
 * Анализ чертежа без проверки правил: детекция → дедупликация → OCR → граф.
 * Асинхронность — только из-за WASM-стадий; DOM не используется.
 */
export async function analyzeImage(image: RawImage, deps?: PipelineDeps): Promise<AnalyzeResult> {
  const detect = deps?.detectSegments ?? detectSegments;
  const recognize = deps?.recognizeLabels ?? recognizeLabels;
  const t0 = performance.now();
  const detected = await detect(image);
  const detectMs = performance.now() - t0;
  const segments = deduplicateSegments(detected);
  if (segments.length === 0) {
    return {
      segments: [], labels: [], vertices: [], graph: {}, unboundLabels: [],
      timings: { detect: detectMs, recognize: 0, graph: 0, verify: 0, total: performance.now() - t0 },
    };
  }
  const t1 = performance.now();
  const labels = await recognize(image);
  const recognizeMs = performance.now() - t1;
  const t2 = performance.now();
  const graphResult = buildGraph(segments, labels);
  const graphMs = performance.now() - t2;
  return {
    segments,
    labels,
    vertices: graphResult.vertices,
    graph: graphResult.graph,
    unboundLabels: graphResult.unboundLabels,
    timings: {
      detect: detectMs, recognize: recognizeMs, graph: graphMs, verify: 0, total: performance.now() - t0,
    },
  };
}
```

and shrink `analyzeDrawing` to:

```ts
export async function analyzeDrawing(
  image: RawImage,
  rule: Rule,
  epsilon: number,
  deps?: PipelineDeps,
): Promise<PipelineResult> {
  const analysis = await analyzeImage(image, deps);
  const t3 = performance.now();
  const verdict = verify({ graph: analysis.graph, rule, epsilon });
  const verifyMs = performance.now() - t3;
  const emptyVerdict: VerifyResult = {
    status: 'Error',
    message: '[Status: Error] На чертеже не найдено отрезков',
    epsilon,
  };
  return {
    ...analysis,
    verdict: analysis.segments.length === 0 ? emptyVerdict : verdict,
    timings: { ...analysis.timings, verify: verifyMs, total: analysis.timings.total + verifyMs },
  };
}
```

  key.type = 'password';
  key.autocomplete = 'off';
  const button = document.createElement('button');
  button.type = 'button';
  button.id = 'gemini-extract';
  button.className = 'btn-primary';
  button.textContent = 'Уточнить через ИИ';
  const error = document.createElement('p');
  error.className = 'fallback-error';

  field.append(caption, key);
  details.append(summary, hint, field, button, error);

  key.addEventListener('input', () => {
    try {
      if (key.value.length > 0) localStorage.setItem(STORAGE_KEY, key.value);
      else localStorage.removeItem(STORAGE_KEY);
    } catch {
      /* localStorage недоступен — ключ живёт в поле до перезагрузки */
    }
  });
  button.addEventListener('click', () => {
    if (key.value.trim().length > 0) onExtract(key.value.trim());
  });
  try {
    key.value = localStorage.getItem(STORAGE_KEY) ?? '';
  } catch {
    /* см. выше */
  }

  return {
    root: details,
    setOpen: (open) => {
      details.open = open;
    },
    setError: (message) => {
      error.textContent = message ?? '';
    },
    setBusy: (busy) => {
      button.disabled = busy;
      button.textContent = busy ? 'Запрос…' : 'Уточнить через ИИ';
    },
  };
}

  const region = card.querySelector('#verdict-region');

    rule.textContent = formatRelation(relation);
    const message = document.createElement('span');
    message.className = 'check-message';
    message.textContent = result.message;
    row.append(createVerdictBadge(result.status), rule, message);
    children.push(row);
  }
  const epsilon = evaluation.results[0]?.result.epsilon;
  if (epsilon !== undefined) {
    children.push(paragraph('verdict-meta', `ε = ${formatEpsilon(epsilon)}`));
  }
  if (timing) children.push(paragraph('verdict-timing mono', timing));
  if (notes.length > 0) {
    const wrap = document.createElement('div');
    wrap.className = 'verdict-notes';
    for (const note of notes) wrap.append(paragraph('verdict-note', note));
    children.push(wrap);
  }
  region.replaceChildren(...children);
}

function paragraph(className: string, text: string): HTMLParagraphElement {
  const el = document.createElement('p');
  el.className = className;
  el.textContent = text;
  return el;
}

  /** Демо-режим: мгновенный evaluateRules на графе демо-сцены (без CV/OCR). */

    if (!state.raw || state.analyzing || !state.confirmed) return;
    setBusy(true);
    setChecklistPlaceholder(checklistCard, 'Анализ чертежа…');
    try {
      const analysis: AnalyzeResult = await analyzeImage(state.raw);
      state.analysis = analysis;
      refreshCanvas();
      applyEvaluation(
        evaluateRules(analysis.graph, state.confirmed.relations, state.epsilon),
        softNotes(analysis.unboundLabels),

      console.error(

        error instanceof Error ? error.stack : error,
      );
      setChecklistPlaceholder(checklistCard, '[Status: Error] Не удалось обработать изображение');
      updateStatusDot('Error');
    } finally {
      setBusy(false);
    }
  }

    const confirm = preview.root.querySelector<HTMLButtonElement>('#confirm-rules');
    if (confirm) confirm.disabled = busy;
  }

  /** Софт-ноты: метки, не привязанные к вершинам (ТЗ §2.5 «иначе drop»). */
  function softNotes(unbound: readonly { char: string }[]): string[] {

  }

    evaluation: RulesEvaluation,
    notes: readonly string[],
    timing?: string,
  ): void {
    state.lastEvaluation = evaluation;
    updateChecklist(checklistCard, evaluation, notes, timing);
    updateStatusDot(evaluation.verdict);
  }

  function updateStatusDot(status: RulesEvaluation['verdict'] | null): void {
    statusDot.className =
      status === 'Success' ? 'status-dot status-dot-ok' : 'status-dot status-dot-danger';
  }

  function onEpsilonChange(epsilon: number): void {
    state.epsilon = epsilon;

    }
  }
